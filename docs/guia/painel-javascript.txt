// Handles headers and JPEGs split at arbitrary network packet boundaries.
class CameraFrames {
  constructor(){this.bytes=new Uint8Array();this.expected=null;}
  push(chunk){
    this.bytes=joinBytes(this.bytes,chunk);
    if(this.bytes.length>524288)throw Error('Quadro excedeu o limite de memória');
    const frames=[];
    for(;;){
      if(this.expected===null){
        let end=-1;
        for(let i=0;i+3<this.bytes.length;i++)if(this.bytes[i]===13&&this.bytes[i+1]===10&&this.bytes[i+2]===13&&this.bytes[i+3]===10){end=i;break;}
        if(end<0){if(this.bytes.length>2048)throw Error('Cabeçalho de vídeo inválido');break;}
        const header=new TextDecoder().decode(this.bytes.subarray(0,end));
        const length=/Content-Length:\s*(\d+)/i.exec(header);
        if(!length||!/^image\/jpeg$/im.test((/Content-Type:\s*([^\r\n]+)/i.exec(header)||[])[1]||''))throw Error('Formato de vídeo inválido');
        this.expected=Number(length[1]);
        if(this.expected<4||this.expected>262144)throw Error('Tamanho de imagem inválido');
        this.bytes=this.bytes.slice(end+4);
      }
      if(this.bytes.length<this.expected)break;
      const frame=this.bytes.slice(0,this.expected);
      this.bytes=this.bytes.slice(this.expected);this.expected=null;
      if(frame[0]!==255||frame[1]!==216)throw Error('Imagem JPEG inválida');
      frames.push(frame);
    }
    return frames;
  }
}

function jpegBackupCheck(bytes){
 if(bytes.length<4||bytes.length>262144||bytes[0]!==255||bytes[1]!==216||bytes.at(-2)!==255||bytes.at(-1)!==217)throw Error('Foto JPEG inválida.');
 let at=2;
 while(at+3<bytes.length){
  if(bytes[at++]!==255)throw Error('Cabeçalho da foto inválido.');
  while(bytes[at]===255)at++;
  const marker=bytes[at++];if(marker===0xda||marker===0xd9)break;
  if(marker===0x01||(marker>=0xd0&&marker<=0xd7))continue;
  if(at+1>=bytes.length)break;
  const size=bytes[at]*256+bytes[at+1];if(size<2||at+size>bytes.length)throw Error('Foto truncada.');
  if([0xc0,0xc1,0xc2,0xc3,0xc5,0xc6,0xc7,0xc9,0xca,0xcb,0xcd,0xce,0xcf].includes(marker)){
   if(size<8)throw Error('Dimensões da foto inválidas.');
   const h=bytes[at+3]*256+bytes[at+4],w=bytes[at+5]*256+bytes[at+6];
   if(!w||!h||w>2048||h>2048||w*h>2000000)throw Error('Foto maior que o tamanho aceito.');return;
  }
  at+=size;
 }
 throw Error('Dimensões da foto ausentes.');
}

async function parseBackup(file){

  if(albums.has(p.key)||!Array.isArray(p.frames)||!p.frames.length||p.frames.length>(event.tipo==='foto_manual'?1:MAX_ALBUM))throw Error('Sequência inválida no arquivo.');
